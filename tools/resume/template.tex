%----------DESIGN TOKENS----------
% Spacing scale: every vertical gap in the document is one of these.
\newlength{\gapXS}\setlength{\gapXS}{1pt}
\newlength{\gapS} \setlength{\gapS}{3pt}
\newlength{\gapM} \setlength{\gapM}{5.5pt}
\newlength{\gapL} \setlength{\gapL}{7pt}

% Type scale
\newcommand{\sizeName}{\fontsize{14.5}{17}\selectfont}
\newcommand{\sizeContact}{\fontsize{10}{12.5}\selectfont}
% body = \normalsize (10pt class x 1.05 carlito scale = 10.5pt)

% Width of the right-hand meta column (dates, locations)
\newlength{\metawidth}\setlength{\metawidth}{1.6in}

% Inline separator. \allowbreak makes it the only legal break point in the
% contact line, so URLs wrap between items instead of mid-token.
\newcommand{\sep}{~\textbullet~\allowbreak}


%----------SECTION HEADS----------
% Centered, bold, title case -- the Harvard OCS convention.
\titleformat{\section}[block]
  {\centering\bfseries\normalsize}
  {}{0em}{}
\titlespacing*{\section}{0pt}{\gapL}{\gapS}


%----------ENTRY BLOCKS----------
% Two-line entry: bold heading + meta on the first line, detail line +
% meta on the second. Built from parboxes, not tabulars, so the text
% layer extracts linearly.
% \leavevmode is load-bearing: a \parbox[t] whose first content is a
% whatsit aligns on that instead of the first text baseline.
\newcommand{\entryline}[3]{%
  \noindent\makebox[\linewidth]{%
    \parbox[t]{\dimexpr\linewidth-\metawidth-\gapS\relax}{\raggedright\leavevmode #1}%
    \hfill
    \parbox[t]{\metawidth}{\raggedleft\leavevmode #2}%
  }\par\vspace{#3}%
}

% #1 heading  #2 meta (location)  #3 detail line  #4 meta (dates)
\newcommand{\entry}[4]{%
  \addvspace{\gapM}%
  \entryline{\bfseries #1}{#2}{\gapXS}%
  \entryline{#3}{#4}{0pt}%
}

% Single-line entry, for blocks with no detail line.
\newcommand{\entryone}[2]{%
  \addvspace{\gapM}%
  \entryline{\bfseries #1}{#2}{0pt}%
}

% Link marker, set inline right after an entry heading. The heading itself
% carries the \href; this names where it points. Nobody retypes a URL from
% a page, so the destination is worth a word and not worth a line: naming
% it costs no vertical space, where the old printed URL cost one line per
% linked entry. Unbolded and small so it reads as a reference, not title.
\newcommand{\entrymarker}[1]{\textnormal{\footnotesize~[#1]}}


%----------BULLETS----------
\newlist{points}{itemize}{1}
\setlist[points]{
  leftmargin=1.5em,
  labelsep=0.6em,
  label={\raisebox{0.15ex}{\footnotesize\textbullet}},
  topsep=3pt,
  itemsep=2pt,
  parsep=0pt,
  partopsep=0pt,
}


%----------SKILLS ROWS----------
% Bold inline label, running text after it -- no aligned label column.
\newcommand{\skillrow}[2]{%
  \noindent\textbf{#1:} #2\par\vspace{1pt}%
}
